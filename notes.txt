● [M] Images pushed to your cloud provider's container registry; lifecycle/cleanup policy 
retaining only the last 10 images per repository 

resource "aws_ecr_lifecycle_policy" "cloudmart" {
  for_each   = aws_ecr_repository.cloudmart
  repository = each.value.name

  policy = jsonencode({
    rules = [
      {
        rulePriority = 1
        description  = "Keep only the last 10 images"
        selection = {
          tagStatus   = "any"
          countType   = "imageCountMoreThan"
          countNumber = 10
        }
        action = {
          type = "expire"
        }
      }
    ]
  })
}
